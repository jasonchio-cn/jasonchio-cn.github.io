  db:
    image: mariadb
    volumes:
      - database:/var/lib/mysql:rw
    restart: always
    networks:
      - private
    environment:

      MYSQL_DATABASE: chevereto
      MYSQL_USER: jason
      MYSQL_PASSWORD: xxx

  chevereto:
    depends_on:
      - db
    image: nmtan/chevereto
    restart: always
    networks:
      - private
    environment:
      CHEVERETO_DB_HOST: db
      CHEVERETO_DB_USERNAME: jason
      CHEVERETO_DB_PASSWORD: xxx
      CHEVERETO_DB_NAME: chevereto
      CHEVERETO_DB_PREFIX: chv_
    volumes:
      - /home/Jason/chv_images:/var/www/html/images:rw  #将照片存放路径映射到了本地存储中
    ports:
      - 8080:80

networks:
  private:
volumes:
  database:
  chevereto_images:

然后docker-compose up -d就可以启动，通过8080端口访问chevereto。

复制到服务器本地路径下

docker cp  containerID:/var/www/html/.htaccess /volume1/docker/

vi /volume1/docker/.htaccess

# Disable server signature
ServerSignature Off

# Disable directory listing (-indexes), Multiviews (-MultiViews) and enable Follow system links (+FollowSymLinks)
Options -Indexes
Options -MultiViews

<IfModule mod_rewrite.c>

    RewriteEngine On

    ...

    RewriteRule . index.php [L]
    #修改上传文件大小增加以下 配置  最大支持 32M 根据自己情况配置
    php_value post_max_size 64M
    php_value upload_max_filesize 32M

</IfModule>

复制到容器目录里面

1

docker cp /volume1/docker/.htaccess containerID:/var/www/html/

修改上传API

1

docker cp containerID:/var/www/html/app/routes/route.api.php ./

将PHP文件复制出来，相应位置增加如下代码

// 更改后的代码如下
$uploaded_id = CHV\Image::uploadToWebsite($source, $user, array('album_id'=>$album));

将修改后的文件复制到容器中替换

1

docker cp ./route.api.php  containerID:/var/www/html/app/routes/route.api.php

配置Typora和PicGo

在PicGo中下载插件web-uploader，填入如下配置

  • API地址：http或https(根据自身情况)://域名/api/1/upload
  • POST参数名：source
  • JSON路径：image.url
  • 自定义Body：{"key":"your key", "user":"username", "album":"albumid"}

key在用户名–>仪表盘–>设置–>API可以找到

username就在chevereto网站右上角

albumid在用户的相册详细信息中可以找到。

配置完成后设置为默认图床，进行上传测试。
